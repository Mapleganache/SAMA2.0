def read_text(relative_path: str) -> str:
    return (WORKSPACE / relative_path).read_text(encoding="utf-8-sig")


def canonical_address(area: str, number: str, bit: str | None) -> str:
    base = f"{area.upper()}{int(number)}"
    return f"{base}.{int(bit)}" if bit is not None else base


def addresses_in(text: str) -> list[str]:
    seen: set[str] = set()
    result: list[str] = []
    for match in ADDRESS_RE.finditer(text):
        address = canonical_address(match.group(1), match.group(2), match.group(3))
        if address not in seen:
            seen.add(address)
            result.append(address)
    return result


def markdown_cell(value: str) -> str:
    return re.sub(r"[*`_]", "", value.strip())


def split_sections(block: str) -> dict[str, str]:
    matches = list(re.finditer(r"^###\s+(.+?)\s*$", block, re.M))
    result: dict[str, str] = {}
    for index, match in enumerate(matches):
        end = matches[index + 1].start() if index + 1 < len(matches) else len(block)
        result[match.group(1).strip()] = block[match.end():end].strip()
    return result

def table_value(block: str, labels: tuple[str, ...]) -> str | None:
    for line in block.splitlines():

        if len(cells) >= 2 and cells[0] in labels:

def parse_manual_signals(source_path: str) -> dict[str, dict]:
    text = read_text(source_path)
    signal_data: dict[str, dict] = defaultdict(lambda: {"symbols": set(), "comments": [], "source_path": source_path})
    current_network: str | None = None

    for line in text.splitlines():
        network_match = re.match(r"^###\s+(N\d+)\s+(.+?)\s*$", line)
        if network_match:
            current_network = network_match.group(1)
            continue
        if not line.startswith("|"):
            continue
        cells = [markdown_cell(cell) for cell in line.strip().strip("|").split("|")]
        if len(cells) == 3 and ADDRESS_RE.fullmatch(cells[0]):
            match = ADDRESS_RE.fullmatch(cells[0])
            assert match is not None
            address = canonical_address(match.group(1), match.group(2), match.group(3))
            if cells[1]:
                signal_data[address]["symbols"].add(cells[1])
            if cells[2]:
                signal_data[address]["comments"].append((cells[2], "符号一览", None))
        elif len(cells) >= 4:
            for match in ADDRESS_RE.finditer(cells[2]):
                address = canonical_address(match.group(1), match.group(2), match.group(3))
                comment = cells[3].strip()
                instruction = cells[1].strip()
                if comment and comment not in ("含义", "---"):
                    signal_data[address]["comments"].append((comment, current_network, instruction))
    return signal_data

def validate(connection: sqlite3.Connection) -> None:
    expected = {
        "equipment": 2,
        "cases": 2,
        "case_sections": 22,
        "issue_findings": 41,
    }
    for table, count in expected.items():
        actual = connection.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]
        if actual != count:
            raise RuntimeError(f"{table}: expected {count}, found {actual}")

    unlinked = connection.execute(
        """
        SELECT COUNT(*) FROM issue_findings AS i
        WHERE i.original_code LIKE 'A-%'
          AND NOT EXISTS (SELECT 1 FROM issue_signals AS x WHERE x.issue_id = i.issue_id)
        """
    ).fetchone()[0]
    if unlinked:
        raise RuntimeError(f"{unlinked} address-logic findings have no linked address")

    result = connection.execute("PRAGMA foreign_key_check").fetchall()
    if result:
        raise RuntimeError(f"foreign key check failed: {result}")

CREATE TABLE IF NOT EXISTS meta (
    key TEXT PRIMARY KEY,
    value TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS equipment (
    equipment_id TEXT PRIMARY KEY,
    brand_zh TEXT NOT NULL,
    brand_en TEXT,
    model TEXT NOT NULL,
    machine_category TEXT,
    cnc_system TEXT,
    plc_pmc TEXT,
    axes TEXT,
    equipment_status TEXT NOT NULL DEFAULT '待现场确认',
    source_basis TEXT
);

CREATE TABLE IF NOT EXISTS cases (
    case_id TEXT PRIMARY KEY,
    equipment_id TEXT NOT NULL REFERENCES equipment(equipment_id),
    title TEXT NOT NULL,
    case_date TEXT,
    case_type TEXT NOT NULL,
    record_status TEXT NOT NULL,
    verification_status TEXT NOT NULL,
    overall_summary TEXT,
    case_file TEXT NOT NULL,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS case_sections (
    case_id TEXT NOT NULL REFERENCES cases(case_id) ON DELETE CASCADE,
    section_code TEXT NOT NULL,
    phase TEXT NOT NULL,
    domain TEXT NOT NULL,
    ordinal INTEGER NOT NULL,
    content TEXT,
    completion_status TEXT NOT NULL CHECK (completion_status IN ('未提供', '已有资料', '不适用', '待补充')),
    PRIMARY KEY (case_id, section_code)
);

CREATE TABLE IF NOT EXISTS issue_findings (
    issue_id TEXT PRIMARY KEY,
    case_id TEXT NOT NULL REFERENCES cases(case_id) ON DELETE CASCADE,
    original_code TEXT NOT NULL,
    title TEXT NOT NULL,
    technical_domain TEXT NOT NULL,
    subsystem TEXT NOT NULL,
    finding_type TEXT NOT NULL,
    risk_level TEXT NOT NULL,
    verification_status TEXT NOT NULL,
    program_ref TEXT,
    network_ref TEXT,
    line_ref TEXT,
    existing_logic TEXT,
    problem TEXT,
    impact TEXT,
    recommendation TEXT,
    raw_markdown TEXT NOT NULL,
    source_path TEXT NOT NULL,
    UNIQUE (case_id, original_code)
);

CREATE TABLE IF NOT EXISTS signals (
    signal_id INTEGER PRIMARY KEY AUTOINCREMENT,
    equipment_id TEXT NOT NULL REFERENCES equipment(equipment_id) ON DELETE CASCADE,
    canonical_address TEXT NOT NULL,
    address_area TEXT NOT NULL,
    address_number INTEGER,
    bit_number INTEGER,
    symbol_name TEXT,
    primary_comment TEXT,
    comment_status TEXT NOT NULL CHECK (comment_status IN ('单一注释', '多注释待核实', '缺少注释')),
    source_path TEXT NOT NULL,
    UNIQUE (equipment_id, canonical_address)
);

CREATE TABLE IF NOT EXISTS signal_comments (
    signal_comment_id INTEGER PRIMARY KEY AUTOINCREMENT,
    signal_id INTEGER NOT NULL REFERENCES signals(signal_id) ON DELETE CASCADE,
    comment_text TEXT NOT NULL,
    network_ref TEXT,
    instruction TEXT,
    source_path TEXT NOT NULL,
    UNIQUE (signal_id, comment_text, network_ref, instruction, source_path)
);

CREATE TABLE IF NOT EXISTS issue_signals (
    issue_id TEXT NOT NULL REFERENCES issue_findings(issue_id) ON DELETE CASCADE,
    signal_id INTEGER NOT NULL REFERENCES signals(signal_id) ON DELETE CASCADE,
    role TEXT NOT NULL DEFAULT '涉及地址',
    comment_at_analysis TEXT,
    evidence_status TEXT NOT NULL DEFAULT '地址与注释联合待核实',
    PRIMARY KEY (issue_id, signal_id)
);

CREATE TABLE IF NOT EXISTS source_documents (
    source_id TEXT PRIMARY KEY,
    equipment_id TEXT REFERENCES equipment(equipment_id),
    name TEXT NOT NULL,
    path TEXT NOT NULL UNIQUE,
    source_type TEXT NOT NULL,
    technical_domain TEXT NOT NULL,
    scope TEXT NOT NULL,
    authority_level TEXT NOT NULL,
    verification_status TEXT NOT NULL,
    notes TEXT
);

CREATE TABLE IF NOT EXISTS case_sources (
    case_id TEXT NOT NULL REFERENCES cases(case_id) ON DELETE CASCADE,
    source_id TEXT NOT NULL REFERENCES source_documents(source_id),
    evidence_role TEXT NOT NULL,
    PRIMARY KEY (case_id, source_id)
);

CREATE VIEW IF NOT EXISTS v_issue_signal_context AS
SELECT
    i.issue_id,
    i.title AS issue_title,
    i.risk_level,
    i.verification_status,
    s.canonical_address,
    s.symbol_name,
    s.primary_comment,
    s.comment_status,
    x.role,
    x.comment_at_analysis,
    x.evidence_status
FROM issue_findings AS i
JOIN issue_signals AS x ON x.issue_id = i.issue_id
JOIN signals AS s ON s.signal_id = x.signal_id;

CREATE VIEW IF NOT EXISTS v_case_completeness AS
SELECT
    c.case_id,
    c.title,
    COUNT(cs.section_code) AS required_sections,
    SUM(CASE WHEN cs.completion_status = '已有资料' THEN 1 ELSE 0 END) AS completed_sections,
    SUM(CASE WHEN cs.completion_status IN ('未提供', '待补充') THEN 1 ELSE 0 END) AS incomplete_sections
FROM cases AS c
LEFT JOIN case_sections AS cs ON cs.case_id = c.case_id
GROUP BY c.case_id, c.title;

CREATE INDEX IF NOT EXISTS idx_issue_case ON issue_findings(case_id);
CREATE INDEX IF NOT EXISTS idx_issue_subsystem ON issue_findings(subsystem, risk_level);
CREATE INDEX IF NOT EXISTS idx_signal_address ON signals(equipment_id, canonical_address);
CREATE INDEX IF NOT EXISTS idx_source_equipment ON source_documents(equipment_id);
